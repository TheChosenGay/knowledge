## 1. 因子计算公式

### 动量因子（跳过最近1个月）

```python
# 252日动量 = 一个月前价格 / 一年前价格 - 1
momentum = close.shift(21) / close.shift(252) - 1
```

### 反转因子（短期跌幅）

```python
# 5日反转：过去5天跌了多少
reversal_5 = close / close.shift(5) - 1

# 20日反转
reversal_20 = close / close.shift(20) - 1
```

### 波动率因子

```python
# 20日波动率 = 日收益率标准差
returns = close.pct_change()
volatility_20 = returns.rolling(20).std()
```

### Amihud 非流动性指标

```python
# 衡量每单位成交额引起的价格变化
# 值越大 = 流动性越差
illiquidity = (abs(returns) / amount).rolling(20).mean()
```

### 振幅因子

```python
# 20日平均日内振幅
intraday_range = (high - low) / close
high_low_ratio = intraday_range.rolling(20).mean()
```

## 2. A股真实费率计算

```python
def calculate_fees(side, gross_amount):
    """
    A股散户真实交易成本
    side: 'buy' | 'sell'
    """
    commission = max(gross_amount * 0.0003, 5.0)   # 佣金万三,最低5元
    stamp_tax = gross_amount * 0.001 if side == 'sell' else 0  # 印花税千一,仅卖出
    transfer_fee = gross_amount * 0.00001           # 过户费十万分之一
    total = commission + stamp_tax + transfer_fee
    return total

# 单边成本 ≈ 0.18%（含5bps滑点）
# 双边一次完整买卖 ≈ 0.36%
```

## 3. 滑点计算

```python
def apply_slippage(side, price, slippage_bps=5):
    """买入滑点向上，卖出滑点向下"""
    if side == 'buy':
        return price * (1 + slippage_bps / 10000)
    else:
        return price * (1 - slippage_bps / 10000)
```

## 4. 多因子打分选股

```python
def score_stocks(factor_df, config):
    """
    多因子加权打分
    factor_df: 包含各股票各因子z-score的DataFrame
    config: 因子权重配置
    """
    score = pd.Series(0, index=factor_df.index)
    
    for factor in config['factors']:
        zscore = factor_df[factor['name']]  # 已标准化的因子值
        weight = factor['weight']
        direction = factor['direction']     # 1=做多高值, -1=做多低值
        score += zscore * weight * direction
    
    # 按得分排名，取前N只
    return score.rank(ascending=False).nsmallest(top_n)
```

## 5. 调仓频率代码

```python
def get_rebalance_dates(trade_dates, frequency):
    """
    根据调仓频率确定调仓日期
    信号在调仓日收盘产生，次日开盘执行
    """
    if frequency == 'daily':
        return trade_dates  # 每天都是调仓日
    
    elif frequency == 'weekly':
        # 每周最后一个交易日
        return trade_dates.groupby(
            [trade_dates.dt.year, trade_dates.dt.isocalendar().week]
        ).max()
    
    elif frequency == 'monthly':
        # 每月最后一个交易日
        return trade_dates.groupby(
            trade_dates.dt.to_period('M')
        ).max()
```

## 6. 动态回撤风控

```python
# 风控阶段配置
RISK_STAGES = [
    (0.10, 0.70),  # 回撤10% → 仓位降至70%
    (0.20, 0.40),  # 回撤20% → 仓位降至40%
    (0.30, 0.05),  # 回撤30% → 仓位降至5%（近乎清仓）
]

def get_exposure(drawdown):
    """根据当前回撤确定目标仓位"""
    exposure = 1.0
    for threshold, target in sorted(RISK_STAGES):
        if abs(drawdown) >= threshold:
            exposure = target
    return exposure

def apply_exposure(targets, exposure):
    """将目标仓位乘以风控系数"""
    return {stock: weight * exposure 
            for stock, weight in targets.items()}
```

## 8. 涨跌停判断

```python
def is_blocked_by_limit(side, open_price, prev_close):
    """
    判断是否因涨跌停无法成交
    主板10%, 创业板/科创板20%, 北交所30%, ST 5%
    """
    limit_pct = infer_limit_pct(stock_code)  # 根据代码判断
    limit_up = prev_close * (1 + limit_pct)
    limit_down = prev_close * (1 - limit_pct)
    
    if side == 'buy' and open_price >= limit_up:
        return True   # 涨停买不到
    if side == 'sell' and open_price <= limit_down:
        return True   # 跌停卖不掉
    return False
```
